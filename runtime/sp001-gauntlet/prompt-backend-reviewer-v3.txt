type GuidePageProps = {

// The directory the guides ship into (co-located static content, _content convention).
function guidesContentDir(): string {
  return path.join(process.cwd(), "src", "app", "admin", "guides", "_content");
}

// Allowlist of referenceable guide basenames — derived from _guides/registry.json,
// the SAME source scripts/scaffold/readiness-report.js#loadGuideFiles reads. We strip
// path separators exactly as the producer does (.split(/[\\/]/).pop()).
function allowlistedGuideBasenames(): Set<string> {
  const abs = path.join(process.cwd(), "_guides", "registry.json");
  const set = new Set<string>();
  try {
    const reg = JSON.parse(fs.readFileSync(abs, "utf8")) as {
      guides?: Array<{ file?: string; path?: string }>;
    };
    for (const g of reg.guides || []) {
      const f = g.file || g.path;
      if (f) {
        const base = String(f).split(/[\\/]/).pop();
        if (base) set.add(base.replace(/\.md$/i, ""));
      }
    }
  } catch {
    /* registry missing/unreadable → empty allowlist → every ref 404s (fail-closed) */
  }
  return set;
}

// Resolve a requested ref to a SAFE basename, or null. REJECTS any traversal-shaped ref
// OUTRIGHT (path separator or "..") BEFORE the allowlist — even when the basename tail is
// allowlisted — so BOTH ../../etc/passwd AND ../../DEV_SETUP_GUIDE.md 404 (no surprising
// strip-then-allow). Then requires membership in the registry allowlist. Returns null →
// caller 404s, NO disk read of the target ever happens for a non-allowlisted ref.
function resolveGuideRef(rawRef: string): string | null {
  if (!rawRef) return null;
  // Reject traversal-SHAPED refs outright, before any stripping or allowlist check.
  if (/[\\/]/.test(rawRef) || rawRef.includes("..")) return null;
  // No separators remain — strip only a trailing .md the caller may have appended.
  const base = String(rawRef).replace(/\.md$/i, "");
  // Defense in depth: reject hidden / empty leftovers.
  if (!base || base === "." || base === "..") return null;
  const allow = allowlistedGuideBasenames();
  return allow.has(base) ? base : null;
}

import path from "node:path";
import { revalidatePath } from "next/cache";
import { resolveAdminActor } from "@/lib/admin/config";
import { patchChecklistItem } from "@/lib/readiness/writeback";

// Result the client can surface — we never throw-and-swallow. A denied gate, an
// un-toggleable item, or a missing id all come back as { ok: false } with a
// reason the panel can render.
export type ToggleReadinessResult =
  | { ok: true; id: string; checked: boolean }
  | { ok: false; error: string };

// FOUNDERS_CHECKLIST.md is the source of truth (no server-persisted check state).
// It lives at the product repo root; resolve it from cwd the same way the
// producer (scripts/scaffold/readiness-report.js) reads it.
function checklistPath(): string {
  return path.join(process.cwd(), "FOUNDERS_CHECKLIST.md");
}

function formValue(formData: FormData, name: string): string {
  const value = formData.get(name);
  return typeof value === "string" ? value.trim() : "";
}

// Derive an item's owner_class SERVER-SIDE from the readiness producer (the source
// of truth). The authorization decision must NOT depend on client form input — a
// client cannot relabel a sprint-work/waiver gate as "owner-action" to force a
// phantom patch. Mirrors the page's loadReport() require pattern (CJS dev-tooling
// module at the product repo root; resolved at request time, server-only).
// Fail-closed: any producer/lookup failure returns null → the toggle is rejected.
function resolveOwnerClass(id: string): string | null {
  try {
    const p = require("node:path") as typeof import("node:path");
    const producerPath = p.join(
      process.cwd(),
      "scripts",
      "scaffold",
      "readiness-report.js",
    );
    // eslint-disable-next-line @typescript-eslint/no-var-requires
    const { buildReadinessReport } = require(producerPath);
    const report = buildReadinessReport(process.cwd(), {
      generated_at: new Date().toISOString(),
    }) as { items?: Array<{ id: string; owner_class: string }> };
    const found = (report.items || []).find((it) => it.id === id);
    return found ? found.owner_class : null;
  } catch {
    return null;
  }
}

export async function toggleReadinessItemAction(
  formData: FormData,
): Promise<ToggleReadinessResult> {
  // GATE: same founder check as the admin route — allowlist is the first gate,
  // default-deny. (resolveAdminActor + !actor.allowed; the prod dev-email
  // fallback fails closed inside resolveAdminActor.)
  const actor = await resolveAdminActor();
  if (!actor.allowed) {
    return { ok: false, error: "Admin access denied: founder not allowlisted" };
  }

  const id = formValue(formData, "id");
  if (!id) {
    return { ok: false, error: "Missing readiness item id" };
  }

  // Only owner-action items map to a toggleable FOUNDERS_CHECKLIST.md line.
  // SECURITY: owner_class is derived SERVER-SIDE (resolveOwnerClass) from the
  // producer, never trusted from client form input. waiver / sprint-work gates are
  // injected (no .md backing) and are not founder-toggleable — reject rather than
  // attempt a phantom patch; an unknown id (absent from producer output) is rejected.
  const ownerClass = resolveOwnerClass(id);
  if (ownerClass === null) {
    return { ok: false, error: "Readiness item not found" };
  }
  if (ownerClass !== "owner-action") {
    return {
      ok: false,
      error: `Item is not founder-toggleable (owner_class=${ownerClass})`,
    };
  }

  const nextChecked = formValue(formData, "nextChecked") === "true";

  const result = patchChecklistItem(checklistPath(), id, nextChecked);
  if (!result.ok) {
    // missing-id / read / write failures surface, never silently no-op.
    return { ok: false, error: result.error };
  }

  revalidatePath("/admin/readiness");
  return { ok: true, id: result.id, checked: result.checked };
}

//
// WHY NOT render-from-model: the producer's parse(file) drops every non-item
// line (human notes, ## headers, schema/declared_stack metadata, both markers)
// and render() regenerates from constants with checks reset — render(parse(x))
// is doubly lossy. Mutating through it would silently destroy human annotations
// and reset every checkbox. So write-back is a NET-NEW byte-preserving patch:
// it touches exactly one line and leaves every other byte untouched.
//
// Contract:
//   patchChecklistItem(checklistPath, id, nextChecked)
//   1. re-reads the file from disk immediately before patching (patch-on-current)
//   2. finds the single line whose `id=<id>` marker matches `id`
//   3. toggles that line's [ ] <-> [x] in place, preserving every other byte
//   4. writes atomically (write tmp, renameSync over original)
//   5. if id not found -> NO write (no tmp left), returns { ok: false }
// ---------------------------------------------------------------------------

export type PatchResult =
  | { ok: true; id: string; checked: boolean; changed: boolean }
  | { ok: false; id: string; error: string };

// A checklist item line, e.g.
//   - [ ] id=provider.accounts dim=product source=core Create ...
//   - [x] id=domain.dns dim=deployment source=core Confirm ...
// We match the checkbox + the `id=` token explicitly so a label that happens to
// contain the same substring can never be mistaken for the item id.
const ID_TOKEN = /(^\s*-\s+\[)([ xX])(\]\s+id=)([a-z0-9_.-]+)/;

function lineMatchesId(line: string, id: string): boolean {
  const m = ID_TOKEN.exec(line);
  return m !== null && m[4] === id;
}

type LineSegment = { content: string; eol: string };

/**
 * Split text into per-line { content, eol } segments WITHOUT normalizing line endings.
 * Each segment carries its OWN original terminator (\r\n, \r, \n, or "" for the final line when
 * the file has no trailing newline). The concatenation of `content + eol` over all segments
 * equals the input byte-for-byte — that exact round-trip is what lets the patcher preserve
 * CRLF/LF/mixed EOLs everywhere except the single glyph it toggles.
 *
 * Implementation: a global match for "run of non-EOL chars, optionally followed by ONE EOL".
 * The regex is anchored to consume the whole string left-to-right with no gaps. We skip the
 * final zero-length match the global form emits at end-of-string so we don't append a phantom
 * empty segment after a trailing newline.
 */
function splitLinesPreservingEol(text: string): LineSegment[] {
  const segments: LineSegment[] = [];
  const re = /([^\r\n]*)(\r\n|\r|\n|$)/g;
  let m: RegExpExecArray | null;
  while ((m = re.exec(text)) !== null) {
    // The trailing `$` alternative matches an empty string at end-of-input AFTER the last real
    // line was already consumed; that yields content="" eol="" with index===text.length. Skip it
    // so a "...\n"-terminated file does not gain a spurious empty segment.
    if (m[0] === "" && m.index === text.length) break;
    segments.push({ content: m[1], eol: m[2] === "" ? "" : m[2] });
    // A zero-width match (only the `$` matched at a non-end position cannot happen, but guard
    // against any zero-length advance to avoid an infinite loop).
    if (m.index === re.lastIndex) re.lastIndex += 1;
  }
  if (segments.length === 0) segments.push({ content: "", eol: "" });
  return segments;
}

/**
 * Toggle a single checklist item's checkbox in place.
 *
 * @param checklistPath absolute path to FOUNDERS_CHECKLIST.md
 * @param id            the item id (matched against the `id=` marker)
 * @param nextChecked   desired checked state (true -> [x], false -> [ ])
 */
export function patchChecklistItem(
  checklistPath: string,
  id: string,
  nextChecked: boolean,
): PatchResult {
  let raw: string;
  try {
    // (1) patch-on-current: read the LATEST bytes from disk, not a stale snapshot.
    raw = readFileSync(checklistPath, "utf8");
  } catch (e) {
    return {
      ok: false,
      id,
      error: `could not read checklist: ${(e as Error).message}`,
    };
  }

  // Split into (content, eol) segments WITHOUT collapsing the line endings. Each segment keeps
  // its OWN original terminator (\r\n, \r, \n, or "" for a final line with no trailing newline),
  // so a CRLF / mixed-EOL file is reproduced byte-for-byte. A naive split(/\r?\n/) + join with a
  // single chosen newline NORMALIZES every EOL and violates AC-A6 byte-preservation on mixed-EOL
  // files — so we never do that.
  const segments = splitLinesPreservingEol(raw);

  // (2) find the single line carrying this id.
  let targetIndex = -1;
  for (let i = 0; i < segments.length; i += 1) {
    if (lineMatchesId(segments[i].content, id)) {
      targetIndex = i;
      break;
    }
  }

  // (5) id not found -> NO write, return error. Never append / render-from-model.
  if (targetIndex === -1) {
    return { ok: false, id, error: `checklist item not found: id=${id}` };
  }

  // (3) toggle ONLY the checkbox glyph on that line's CONTENT; every other byte — including
  // every other line's EOL AND this line's own EOL — is preserved exactly.
  const original = segments[targetIndex].content;
  const desiredGlyph = nextChecked ? "x" : " ";
  const patched = original.replace(
    ID_TOKEN,
    (_full, pre, _glyph, mid, matchedId) =>
      `${pre}${desiredGlyph}${mid}${matchedId}`,
  );

  const changed = patched !== original;
  if (changed) {
    segments[targetIndex].content = patched;
  }

  // Reassemble verbatim: content + its original eol, in order. Non-target bytes are identical to
  // the input, so a raw byte-diff shows ONLY the single toggled glyph.
  const nextRaw = segments.map((s) => s.content + s.eol).join("");

  // (4) atomic write: write a sibling tmp file, then renameSync over the original.
  // renameSync is atomic on the same filesystem, so a reader sees either the old
  // file or the fully-written new file, never a torn write.
  const tmpPath = `${checklistPath}.${process.pid}.${randomBytes(6).toString("hex")}.tmp`;
  try {
    writeFileSync(tmpPath, nextRaw, "utf8");
    renameSync(tmpPath, checklistPath);
  } catch (e) {
    // Cleanup on EVERY failure path so a failed patch never leaves an orphan tmp:
    //   - writeFileSync error: tmp may be partially written or absent.
    //   - renameSync error: tmp is fully written but NOT moved — it MUST be removed here.
    // existsSync-guard + try/catch makes the removal robust on both paths (and on a partial tmp).
    try {
      if (existsSync(tmpPath)) unlinkSync(tmpPath);
    } catch {
      /* last-resort: nothing more we can do; never throw out of cleanup */
    }
    return {
      ok: false,
      id,
      error: `atomic write failed: ${(e as Error).message}`,
    };
  }

  return { ok: true, id, checked: nextChecked, changed };
}

// Readiness grouping + ordering — the UI groups and orders; it does NOT preserve the
// raw producer order (UX spec LIST GROUPING).
//
//   1. "Do this first" = open + owner-action, lead-time items first (longest lead-time
//      / longest real-world pole at the top), then the rest.
//   2. "Blocked"       = status blocked.
//   3. "In the sprint" = open + (sprint-work | waiver) — "we're handling these".
//   4. "Done"          = status done, collapsed by default.
//
// 0-item groups are omitted by the caller. Pure + deterministic — no clock, no I/O.

import type {
  ReadinessGroup,
  ReadinessPhase,
  ReadinessReport,
  ReadinessSummary,
  ReadinessView,
  ReadinessViewItem,
} from "./types";

// Strip the report to the brand-clean view subset (drops schema, product_id, source).
export function toReadinessView(report: ReadinessReport): ReadinessView {
  return {
    composite: report.composite,
    summary: report.summary,
    items: report.items.map((it) => ({
      id: it.id,
      label: it.label,
      dim: it.dim,
      status: it.status,
      owner_class: it.owner_class,
      lead_time: it.lead_time,
      blocker: it.blocker,
      deep_link: it.deep_link,
    })),
  };
}

// Stable sort: lead-time items first (days desc — longest pole on top), then the rest in
// their original relative order. JS Array.sort is stable on modern runtimes (V8/Node 12+).
function orderDoFirst(items: ReadinessViewItem[]): ReadinessViewItem[] {
  return [...items].sort((a, b) => {
    const aLead = a.lead_time ? a.lead_time.days : -1;
    const bLead = b.lead_time ? b.lead_time.days : -1;
    return bLead - aLead;
  });
}

// The id of the single longest-lead-time open owner-action item ("Start now"), or null.
export function longestLeadOpenItemId(view: ReadinessView): string | null {
  let best: ReadinessViewItem | null = null;
  for (const it of view.items) {
    if (it.status !== "open" || it.owner_class !== "owner-action") continue;
    if (!it.lead_time) continue;
    if (!best || !best.lead_time || it.lead_time.days > best.lead_time.days) {
      best = it;
    }
  }
  return best ? best.id : null;
}

export function readinessPhase(summary: ReadinessSummary): ReadinessPhase {
  if (summary.completed === 0) return "cold";
  if (summary.open === 0 && summary.blocked === 0) return "all-done";
  return "warm";
}

// Group + order the view items. Returns ONLY non-empty groups, in display order.
export function groupReadinessItems(view: ReadinessView): ReadinessGroup[] {
  const doFirst: ReadinessViewItem[] = [];
  const blocked: ReadinessViewItem[] = [];
  const inSprint: ReadinessViewItem[] = [];
  const done: ReadinessViewItem[] = [];

  for (const it of view.items) {
    if (it.status === "done") {
      done.push(it);
    } else if (it.status === "blocked") {
      blocked.push(it);
    } else if (it.owner_class === "owner-action") {
      doFirst.push(it);
    } else {
      inSprint.push(it);
    }
  }

  const groups: ReadinessGroup[] = [
    {
      key: "do-first",
      heading: "Do this first",
      items: orderDoFirst(doFirst),
      collapsedByDefault: false,
    },
    { key: "blocked", heading: "Blocked", items: blocked, collapsedByDefault: false },
    {
      key: "in-sprint",
      heading: "In the sprint",
      items: inSprint,
      collapsedByDefault: false,
    },
    {
      key: "done",
      heading: `Done (${done.length})`,
      items: done,
      collapsedByDefault: true,
    },
  ];

  return groups.filter((g) => g.items.length > 0);
}

import { createHmac, timingSafeEqual } from "node:crypto";
import { cookies } from "next/headers";

export type AdminActor = {
  email: string;
  allowed: boolean;
  source: "signed-cookie" | "dev-env" | "missing";
};

export type AdminFeatureFlag = {
  key: string;
  label: string;
  enabled: boolean;
};

export const ADMIN_FEATURE_FLAGS: AdminFeatureFlag[] = [
  {
    key: "admin.project_sections",
    label: "Project sections",
    enabled: true,
  },
  {
    key: "admin.event_feed",
    label: "Event feed",
    enabled: true,
  },
];

const DEFAULT_ADMIN_SESSION_COOKIE = "warpos_admin_session";

function normalizeEmail(email: string): string {
  return email.trim().toLowerCase();
}

function splitEmails(value: string | undefined): string[] {
  return (value || "")
    .split(/[,\s]+/)
    .map(normalizeEmail)
    .filter(Boolean);
}

export function founderEmailAllowlist(): string[] {
  return splitEmails(process.env.ADMIN_FOUNDER_EMAILS);
}

export function adminSessionCookieName(): string {
  return (
    process.env.ADMIN_SESSION_COOKIE || DEFAULT_ADMIN_SESSION_COOKIE
  ).trim();
}

export function isFounderEmailAllowed(email: string | null | undefined): boolean {
  const normalized = normalizeEmail(email || "");
  return Boolean(normalized && founderEmailAllowlist().includes(normalized));
}

function adminSessionSecret(): string {
  return (process.env.ADMIN_SESSION_SECRET || "").trim();
}

function signAdminEmail(email: string, secret: string): string {
  return createHmac("sha256", secret).update(email).digest("base64url");
}

function safeEqual(a: string, b: string): boolean {
  const left = Buffer.from(a);
  const right = Buffer.from(b);
  return left.length === right.length && timingSafeEqual(left, right);
}

export function verifyAdminSessionCookie(value: string | undefined): string | null {
  const secret = adminSessionSecret();
  if (!secret || !value) return null;

  const separatorIndex = value.lastIndexOf(".");
  if (separatorIndex <= 0) return null;

  const encodedEmail = value.slice(0, separatorIndex);
  const signature = value.slice(separatorIndex + 1);
  const email = normalizeEmail(Buffer.from(encodedEmail, "base64url").toString("utf8"));
  if (!email || !signature) return null;

  const expectedSignature = signAdminEmail(email, secret);
  return safeEqual(signature, expectedSignature) ? email : null;
}

export function signAdminSessionEmail(email: string): string | null {
  const secret = adminSessionSecret();
  const normalizedEmail = normalizeEmail(email);
  if (!secret || !normalizedEmail) return null;

  const encodedEmail = Buffer.from(normalizedEmail).toString("base64url");
  return `${encodedEmail}.${signAdminEmail(normalizedEmail, secret)}`;
}

export async function resolveAdminActor(): Promise<AdminActor> {
  const requestCookies = await cookies();
  const sessionEmail = verifyAdminSessionCookie(
    requestCookies.get(adminSessionCookieName())?.value,
  );
  const devEmail =
    process.env.NODE_ENV !== "production"
      ? normalizeEmail(process.env.ADMIN_DEV_EMAIL || "")
      : "";
  const email = sessionEmail || devEmail;
  return {
    email,
    allowed: isFounderEmailAllowed(email),
    source: sessionEmail ? "signed-cookie" : devEmail ? "dev-env" : "missing",
  };
}

export async function requireFounderAdmin(): Promise<AdminActor> {
  const actor = await resolveAdminActor();
  if (!actor.allowed) {
    throw new Error("Admin access denied: founder email is not allowlisted");
  }
  return actor;
}
